Host github.com-vrcx
  HostName github.com
  User git
  IdentityFile ~/.ssh/id_ed25519_github_vrcx
  IdentitiesOnly yes

# 说明：
# 这是要追加到 C:\Users\<你的用户名>\_ssh\config 文件里的内容
# （Windows 上 OpenSSH 的配置路径是 C:\Users\<用户名>\.ssh\config）
#
# 作用是：当 git 用 github.com-vrcx 这个主机名时，强制使用上面那把密钥，
# 而不是电脑上可能存在的其他密钥。
#
# 仓库远程地址因此写成：
#   git@github.com-vrcx:W4nN1ng/VRCX-sweetCandy.git
# 查看当前远程：
#   git -C C:\Users\28041\vrcx-fork remote -v
